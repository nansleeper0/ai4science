import glob
from scipy.signal import find_peaks
from scipy.interpolate import interp1d

def find_peaks_fwhm(filepath):
    """Find peaks and their FWHM in a spectrum file."""
    data = np.loadtxt(filepath, skiprows=1)
    wl = data[:, 0]
    intensity = data[:, 1]
    
    # Find peaks using scipy
    peaks, props = find_peaks(intensity, height=0, distance=10)
    
    results = []
    for pk in peaks:
        peak_height = intensity[pk]
        peak_wl = wl[pk]
        
        # Find FWHM by interpolation
        half_max = peak_height / 2.0
        
        # Search left side
        left_idx = pk
        for i in range(pk, -1, -1):
            if intensity[i] < half_max:
                left_idx = i
                break
        
        # Search right side
        right_idx = pk
        for i in range(pk, len(intensity)):
            if intensity[i] < half_max:
                right_idx = i
                break
        
        # Linear interpolation for more precise FWHM
        if left_idx < pk:
            # Interpolate between left_idx and left_idx+1
            wl_left = wl[left_idx] + (wl[left_idx+1] - wl[left_idx]) * \
                      (half_max - intensity[left_idx]) / (intensity[left_idx+1] - intensity[left_idx])
        else:
            wl_left = wl[pk]
        
        if right_idx > pk:
            wl_right = wl[right_idx-1] + (wl[right_idx] - wl[right_idx-1]) * \
                       (half_max - intensity[right_idx-1]) / (intensity[right_idx] - intensity[right_idx-1])
        else:
            wl_right = wl[pk]
        
        fwhm = wl_right - wl_left
        
        results.append({
            'wavelength': peak_wl,
            'intensity': peak_height,
            'fwhm': fwhm,
            'half_max': half_max
        })
    
    return results

import glob

def find_significant_peaks(filepath, threshold=0.3):
    """Find significant peaks above threshold."""
    data = np.loadtxt(filepath, skiprows=1)
    wl = data[:, 0]
    intensity = data[:, 1]
    
    # Smooth slightly to reduce noise
    from scipy.ndimage import uniform_filter1d
    smoothed = uniform_filter1d(intensity, size=3)
    
    # Find local maxima in smoothed data
    peaks = []
    for i in range(1, len(smoothed) - 1):
        if smoothed[i] > smoothed[i-1] and smoothed[i] > smoothed[i+1] and smoothed[i] > threshold:
            # FWHM via interpolation
            half_max = smoothed[i] / 2.0
            pk = i
            
            # Left
            left_idx = pk
            for j in range(pk, -1, -1):
                if smoothed[j] < half_max:
                    left_idx = j
                    break
            if left_idx < pk:
                wl_left = wl[left_idx] + (wl[left_idx+1] - wl[left_idx]) * \
                          (half_max - smoothed[left_idx]) / (smoothed[left_idx+1] - smoothed[left_idx])
            else:
                wl_left = wl[pk]
            
            # Right
            right_idx = pk
            for j in range(pk, len(smoothed)):
                if smoothed[j] < half_max:
                    right_idx = j
                    break
            if right_idx > pk:
                wl_right = wl[right_idx-1] + (wl[right_idx] - wl[right_idx-1]) * \
                           (half_max - smoothed[right_idx-1]) / (smoothed[right_idx] - smoothed[right_idx-1])
            else:
                wl_right = wl[pk]
            
            fwhm = wl_right - wl_left
            peaks.append({
                'wavelength': wl[pk],
                'intensity': smoothed[pk],
                'fwhm': fwhm
            })
    
    return peaks
